\section{Chemické rovnice}

\subsection{Neredoxní rovnice}

Ukázat algebraický způsob vyčíslování!

\begin{itemize}
	\item \ce{CaCl2 + Na3PO4 -> Ca3(PO4)2 + NaCl}
	\item \ce{3 CaCl2 + 2 Na3PO4 -> Ca3(PO4)2 + 6 NaCl}
	\item \ce{POCl3 + H2O -> H3PO4 + HCl}
	\item \ce{POCl3 + 3 H2O -> H3PO4 + 3 HCl}
	\item \ce{Na2B4O7 + HCl + H2O -> H3BO3 + NaCl}
	\item \ce{Na2B4O7 + 2 HCl + 5 H2O -> 4 H3BO3 + 2 NaCl}
	\item \ce{Al2(SO4)3 + NH3 + H2O -> Al(OH)3 + (NH4)2SO4}
	\item \ce{Al2(SO4)3 + 6 NH3 + 6 H2O -> 2 Al(OH)3 + 3 (NH4)2SO4}
	\item \ce{HClO4 + P4O10 -> Cl2O7 + H3PO4}
	\item \ce{12 HClO4 + P4O10 -> 6 Cl2O7 + 4 H3PO4}
\end{itemize}

\subsection{Redoxní rovnice}

\begin{itemize}
	\item \ce{HgO -> Hg + O2}
	\item \ce{2 HgO -> 2 Hg + O2}
	\item \ce{FeSO4 + K2Cr2O7 + H2SO4 -> Fe2(SO4)3 + Cr2(SO4)3 + K2SO4 + H2O}
	\item \ce{6 FeSO4 + K2Cr2O7 + 7 H2SO4 -> 3 Fe2(SO4)3 + 3 Cr2(SO4)3 + K2SO4 + 7 H2O}
	\item \ce{Cu2O + Cu2S -> Cu + SO2}
	\item \ce{2 Cu2O + Cu2S -> 6 Cu + SO2}
	\item \ce{CrCl3 + F2 -> CrF3 + Cl2}
	\item \ce{2 CrCl3 + 3 F2 -> 2 CrF3 + 3 Cl2}
	\item \ce{NH3 + O2 -> NO + H2O}
	\item \ce{4 NH3 + 5 O2 -> 4 NO + 6 H2O}
	\item \ce{S2O$^{2-}_3$ + I2 -> S4O$^{2-}_6$ + I-}
	\item \ce{2 S2O$^{2-}_3$ + I2 -> S4O$^{2-}_6$ + 2 I-}
	\item \ce{S2O$^{2-}_3$ + Br2 + H2O -> SO$^{2-}_4$ + Br- + H+}
	\item \ce{S2O$^{2-}_3$ + 2 Br2 + 5 H2O -> 2 SO$^{2-}_4$ + 4 Br- + 10 H+}
	\item \ce{HNO2 + H+ + Cr2O$^{2-}_7$ -> NO$_3^-$ + Cr^{3+} + H2O}
	\item \ce{3 HNO2 + 5 H+ + Cr2O$^{2-}_7$ -> 3 NO$_3^-$ + 2 Cr^{3+} + 4 H2O}
\end{itemize}
