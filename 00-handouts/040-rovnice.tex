\subsection{Neredoxní rovnice}

Ukázat algebraický způsob vyčíslování!

\begin{itemize}
	\item \ce{CaCl2 + Na3PO4 -> Ca3(PO4)2 + NaCl (3 + 2 -> 1 + 6)}
	\item \ce{POCl3 + H2O -> H3PO4 + HCl (1 + 3 -> 1 + 3)}
	\item \ce{Na2B4O7 + HCl + H2O -> H3BO3 + NaCl (1 + 2 + 5 -> 4 + 2)}
	\item \ce{NaOH + H3PO4 -> Na3PO4 + H2O (3 + 1 -> 1 + 3)}
	\item \ce{Al2(SO4)3 + NH3 + H2O -> Al(OH)3 + (NH4)2SO4 (1 + 6 + 6 -> 2 + 3)}
	\item \ce{HClO4 + P4O10 -> Cl2O7 + H3PO4 (12 + 1 -> 6 + 4)}
	\item \ce{Mg(NO3)2 + Li -> Mg + LiNO3 (1 + 2 -> 1 + 2)}
\end{itemize}

	\item \ce{Cu2O + Cu2S -> Cu + SO2 (2 + 1 -> 6 + 1)}

	\item \ce{NH3 + O2 -> NO + H2O (4 + 5 -> 4 + 6)}
	\item \ce{(NH4)2Cr2O7 -> Cr2O3 + N2 + H2O (1 -> 1 + 1 + 4)}

	\item \ce{H3PO4 -> H3PO3 + PH3 (4 -> 3 + 1)}
